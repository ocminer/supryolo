cmake_minimum_required(VERSION 3.24)
project(supryolo VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
option(YOLO_CUDA "Build the NVIDIA CUDA backend" ON)
option(YOLO_OPENCL "Build the OpenCL GPU backend" ON)
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
add_library(yolo_core src/crypto.cpp src/coin.cpp src/cpu.cpp src/cpu_avx2.cpp)
target_include_directories(yolo_core PUBLIC include third_party)
target_link_libraries(yolo_core PUBLIC OpenSSL::Crypto Threads::Threads)
add_library(yolo_protocol src/stratum.cpp)
target_link_libraries(yolo_protocol PUBLIC yolo_core)
if(YOLO_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 120 CACHE STRING "CUDA architectures")
  endif()
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 20)
  add_library(yolo_cuda src/cuda/blake2b.cu)
  target_compile_options(yolo_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--ptxas-options=-v>)
else()
  add_library(yolo_cuda src/cuda_stub.cpp)
endif()
target_link_libraries(yolo_cuda PUBLIC yolo_core)
if(YOLO_OPENCL)
find_path(YOLO_OPENCL_INCLUDE_DIR CL/cl.h REQUIRED)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/opencl/blake2b.cl" YOLO_OPENCL_SOURCE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/opencl/blake2b.cl)
configure_file(src/opencl/kernel.hpp.in generated/opencl_kernel.hpp @ONLY)
add_library(yolo_opencl src/opencl/runtime.cpp)
target_include_directories(yolo_opencl PRIVATE ${YOLO_OPENCL_INCLUDE_DIR} "${CMAKE_CURRENT_BINARY_DIR}/generated")
else()
add_library(yolo_opencl src/opencl_stub.cpp)
endif()
target_link_libraries(yolo_opencl PUBLIC yolo_core ${CMAKE_DL_LIBS})
add_library(yolo_devices src/devices.cpp)
target_link_libraries(yolo_devices PUBLIC yolo_cuda yolo_opencl)
add_library(yolo_hardware src/hardware/nvml.cpp src/hardware/amd.cpp)
target_include_directories(yolo_hardware PUBLIC include)
target_link_libraries(yolo_hardware PUBLIC ${CMAKE_DL_LIBS})
add_library(yolo_ui src/ui/dashboard.cpp)
target_include_directories(yolo_ui PUBLIC include)
target_link_libraries(yolo_ui PUBLIC Threads::Threads)
add_library(yolo_network src/miner.cpp src/api.cpp)
if(WIN32)
  target_link_libraries(yolo_network PUBLIC ws2_32)
endif()
target_link_libraries(yolo_network PUBLIC yolo_protocol yolo_devices yolo_hardware yolo_ui OpenSSL::SSL)
add_executable(supryolo src/main.cpp)
target_link_libraries(supryolo PRIVATE yolo_network)
add_executable(yolo_tests tests/test.cpp)
target_link_libraries(yolo_tests PRIVATE yolo_network)
enable_testing()
add_test(NAME core_protocol COMMAND yolo_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND AND NOT WIN32)
  add_test(NAME tui_terminal COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/terminal.py $<TARGET_FILE:supryolo>)
  add_test(NAME stratum_cpu COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/stratum_mock.py $<TARGET_FILE:supryolo> --cpu)
endif()

foreach(target yolo_opencl yolo_devices yolo_core yolo_protocol yolo_network yolo_hardware yolo_ui supryolo yolo_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()

if(NOT WIN32)
add_library(fake_nvml SHARED tests/fakes/nvml.cpp)
set_target_properties(fake_nvml PROPERTIES OUTPUT_NAME nvidia-ml SOVERSION 1 LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-libs")
add_executable(hardware_tests tests/hardware.cpp)
target_link_libraries(hardware_tests PRIVATE yolo_hardware)
add_dependencies(hardware_tests fake_nvml)
add_test(NAME gpu_controls COMMAND hardware_tests)
set_tests_properties(gpu_controls PROPERTIES ENVIRONMENT "LD_LIBRARY_PATH=$<TARGET_FILE_DIR:fake_nvml>")
add_executable(ui_tests tests/ui.cpp)
target_link_libraries(ui_tests PRIVATE yolo_ui)
add_test(NAME tui_render COMMAND ui_tests)

add_executable(opencl_tests tests/opencl.cpp)
target_link_libraries(opencl_tests PRIVATE yolo_devices)
add_executable(amd_tests tests/amd.cpp)
target_link_libraries(amd_tests PRIVATE yolo_hardware)
add_test(NAME amd_sensors COMMAND amd_tests)

endif()
add_executable(api_fixture tests/api_server.cpp)
target_link_libraries(api_fixture PRIVATE yolo_network)
if(Python3_FOUND)
  add_test(NAME monitoring_api COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/api.py $<TARGET_FILE:api_fixture>)
endif()
if(Python3_FOUND AND NOT WIN32)
  add_test(NAME packaging_scripts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/packaging.py)
endif()
